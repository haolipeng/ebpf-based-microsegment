# Task: ProcessMonitor 组件实现

## Description

实现 Agent 端的 ProcessMonitor 组件，负责监听 Task 001 创建的 `process_events` ring buffer，收到进程事件后立即查询进程完整路径（/proc/<pid>/exe），并维护用户态进程缓存（LRU + TTL）。解决 eBPF 层无法存储长路径的限制，提供完整进程信息供流量展示和策略匹配使用。

**核心目标**:
- 监听 `process_events` ring buffer 接收进程事件
- 收到事件后立即查询 `/proc/<pid>/exe` 获取进程完整路径
- 维护用户态 LRU 缓存（20000 条目，TTL 5 分钟）
- 提供查询接口供 FlowCollector 使用（Task 4）
- 实现缓存失效和清理机制（防止 PID 重用问题）

package process

import "time"

// ProcessInfo represents complete process information
type ProcessInfo struct {
    PID         uint32    `json:"pid"`
    Comm        string    `json:"comm"`          // Process command name (16 chars)
    Path        string    `json:"path"`          // Full executable path
    ContainerID string    `json:"container_id"`  // Container ID (if any)
    ExecTime    uint64    `json:"exec_time"`     // Process start timestamp (ns)
    CachedTime  time.Time `json:"cached_time"`   // When this info was cached
    PathResolved bool     `json:"path_resolved"` // Whether path was successfully resolved
}

// ProcessEvent represents the event from eBPF ring buffer

    PID         uint32
    Comm        [16]byte
    ExecTime    uint64
    ContainerID [64]byte

**ProcessMonitor 实现**:
```go
// monitor.go

package process

import (
    "context"
    "fmt"
    "log"
    "os"
    "time"

    "github.com/cilium/ebpf"
    "github.com/cilium/ebpf/ringbuf"
)

type ProcessMonitor struct {
    cache        *ProcessCache
    ringbufReader *ringbuf.Reader
    ctx          context.Context
    cancel       context.CancelFunc
}

func NewProcessMonitor(processEventsMap *ebpf.Map) (*ProcessMonitor, error) {
    // Create ring buffer reader
    reader, err := ringbuf.NewReader(processEventsMap)
    if err != nil {
        return nil, fmt.Errorf("failed to create ringbuf reader: %w", err)
    }

    ctx, cancel := context.WithCancel(context.Background())

    return &ProcessMonitor{
        cache:        NewProcessCache(20000, 5*time.Minute),
        ringbufReader: reader,
        ctx:          ctx,
        cancel:       cancel,
    }, nil
}

func (m *ProcessMonitor) Start() {
    go m.processEventsLoop()
}

func (m *ProcessMonitor) Stop() {
    m.cancel()
    m.ringbufReader.Close()
}

// Main event processing loop
func (m *ProcessMonitor) processEventsLoop() {
    for {
        select {
        case <-m.ctx.Done():
            return
        default:
            record, err := m.ringbufReader.Read()
            if err != nil {
                if err == ringbuf.ErrClosed {
                    return
                }
                log.Printf("Error reading from ringbuf: %v", err)
                continue
            }

            // Parse event
            var event ProcessEvent
            if err := binary.Read(bytes.NewReader(record.RawSample), binary.LittleEndian, &event); err != nil {
                log.Printf("Error parsing event: %v", err)
                continue
            }

            // Handle event
            m.handleProcessEvent(&event)
        }
    }
}

// Handle single process event
func (m *ProcessMonitor) handleProcessEvent(event *ProcessEvent) {
    // Immediately query process path (while process still exists)
    path, err := readProcessPath(event.PID)

    info := ProcessInfo{
        PID:          event.PID,
        Comm:         string(bytes.TrimRight(event.Comm[:], "\x00")),
        ContainerID:  string(bytes.TrimRight(event.ContainerID[:], "\x00")),
        ExecTime:     event.ExecTime,
        CachedTime:   time.Now(),
        PathResolved: err == nil,
    }

    if err == nil {
        info.Path = path
    } else {
        // Path resolution failed (process already exited?)
        log.Printf("Failed to resolve path for PID %d: %v", event.PID, err)
        info.Path = ""  // Mark as unresolved
    }

    // Cache the info
    m.cache.Put(event.PID, &info)
}

// Query API: Get process info by PID
func (m *ProcessMonitor) GetProcessInfo(pid uint32) (*ProcessInfo, bool) {
    return m.cache.Get(pid)
}

// Helper: Read process executable path from /proc
func readProcessPath(pid uint32) (string, error) {
    exePath := fmt.Sprintf("/proc/%d/exe", pid)

    // Use os.Readlink to resolve symlink
    path, err := os.Readlink(exePath)
    if err != nil {
        return "", fmt.Errorf("failed to read symlink: %w", err)
    }

    return path, nil
}
```

**进程缓存实现（LRU + TTL）**:
```go
// cache.go

package process

import (
    "container/list"
    "sync"
    "time"
)

type ProcessCache struct {
    maxEntries int
    ttl        time.Duration
    mu         sync.RWMutex
    cache      map[uint32]*cacheEntry
    lruList    *list.List
}

type cacheEntry struct {
    key        uint32
    info       *ProcessInfo
    expireTime time.Time
    element    *list.Element
}

func NewProcessCache(maxEntries int, ttl time.Duration) *ProcessCache {
    pc := &ProcessCache{
        maxEntries: maxEntries,
        ttl:        ttl,
        cache:      make(map[uint32]*cacheEntry),
        lruList:    list.New(),
    }

    // Start background cleaner
    go pc.cleanupExpiredEntries()

    return pc
}

func (pc *ProcessCache) Put(pid uint32, info *ProcessInfo) {
    pc.mu.Lock()
    defer pc.mu.Unlock()

    // Check if already exists
    if entry, ok := pc.cache[pid]; ok {
        // Update existing entry
        entry.info = info
        entry.expireTime = time.Now().Add(pc.ttl)
        pc.lruList.MoveToFront(entry.element)
        return
    }

    // Evict oldest if cache full
    if pc.lruList.Len() >= pc.maxEntries {
        pc.evictOldest()
    }

    // Add new entry
    entry := &cacheEntry{
        key:        pid,
        info:       info,
        expireTime: time.Now().Add(pc.ttl),
    }
    entry.element = pc.lruList.PushFront(entry)
    pc.cache[pid] = entry
}

func (pc *ProcessCache) Get(pid uint32) (*ProcessInfo, bool) {
    pc.mu.RLock()
    defer pc.mu.RUnlock()

    entry, ok := pc.cache[pid]
    if !ok {
        return nil, false
    }

    // Check if expired
    if time.Now().After(entry.expireTime) {
        return nil, false
    }

    // Move to front (LRU)
    pc.lruList.MoveToFront(entry.element)

    return entry.info, true
}

func (pc *ProcessCache) evictOldest() {
    oldest := pc.lruList.Back()
    if oldest == nil {
        return
    }

    entry := oldest.Value.(*cacheEntry)
    pc.lruList.Remove(oldest)
    delete(pc.cache, entry.key)
}

func (pc *ProcessCache) cleanupExpiredEntries() {
    ticker := time.NewTicker(1 * time.Minute)
    defer ticker.Stop()

    for range ticker.C {
        pc.mu.Lock()
        now := time.Now()

        var toRemove []uint32
        for pid, entry := range pc.cache {
            if now.After(entry.expireTime) {
                toRemove = append(toRemove, pid)
            }
        }

        for _, pid := range toRemove {
            entry := pc.cache[pid]
            pc.lruList.Remove(entry.element)
            delete(pc.cache, pid)
        }

        pc.mu.Unlock()
    }
}
```

### 代码模式

- **并发安全**: 使用 `sync.RWMutex` 保护缓存操作
- **LRU 淘汰**: 使用 `container/list` 实现双向链表
- **TTL 清理**: 后台 goroutine 定期清理过期条目
- **错误处理**: 路径查询失败时标记 `PathResolved=false`，不阻塞流程
- **PID 重用检测**: 通过 `exec_time` 时间戳验证缓存是否过期
