# Task 10: 性能压测与多内核兼容性测试

## Overview

对 LPM Trie 实现进行全面的性能压测和多内核版本兼容性验证,确保在生产环境中能达到性能目标。

## Objective

通过系统的性能基准测试和兼容性验证,验证 LPM Trie 优化的有效性,确保在不同内核版本上的稳定性和一致的性能表现。

## Description

执行完整的性能测试和兼容性验证,包括:

1. **性能压测**:
   - 测试 100、500、1000 条策略下的查询延迟
   - 测试不同 CIDR 前缀长度的影响
   - 测试精确匹配、通配符匹配、LPM 匹配的性能对比

2. **多内核兼容性测试**:
   - Kernel 4.11 (LPM Trie 最小版本)
   - Kernel 5.4 (LTS 版本)
   - Kernel 5.10 (LTS 版本)
   - Kernel 5.15 (最新 LTS 版本)

3. **性能对比分析**:
   - Array Map 线性扫描 vs LPM Trie 查找
   - Phase 1 优化 vs Phase 2 LPM Trie
   - 生成性能对比报告和优化建议

4. **eBPF Verifier 验证**:
   - 确保 eBPF 代码在所有内核版本通过 verifier
   - 记录 verifier 日志和警告
   - 解决任何兼容性问题

```c
// benchmark_lpm_lookup.c - LPM 查找性能基准

#include <uapi/linux/ptrace.h>
#include <net/sock.h>
#include <bcc/proto.h>

// 性能统计结构
struct perf_stats {
    __u64 total_lookups;
    __u64 total_time;    // 总耗时(纳秒)
    __u64 min_time;
    __u64 max_time;
    __u64 hits;
    __u64 misses;
};

// 每条规则的查询时间
BPF_PERF_OUTPUT(events);

struct lookup_event {
    __u64 timestamp;
    __u32 rule_id;
    __u32 src_ip;
    __u32 dst_ip;
    __u32 lookup_time_ns;  // 查询耗时(纳秒)
    __u8 hit;
};

// 统计数据
BPF_ARRAY(stats, struct perf_stats, 1);

// LPM 性能测试入口
int trace_lpm_lookup(struct pt_regs *ctx) {
    __u64 start = bpf_ktime_get_ns();

    // LPM 查询代码
    // ...

    __u64 end = bpf_ktime_get_ns();
    __u64 duration = end - start;

    // 记录事件
    struct lookup_event event = {
        .timestamp = start,
        .lookup_time_ns = duration,
        // ...
    };
    events.perf_submit(ctx, &event, sizeof(event));

    return 0;
}
```

#### 测试脚本 (Bash + Python)

```bash
#!/bin/bash
# load_test.sh - 性能压测脚本

set -e

# 配置
POLICY_COUNTS=(100 500 1000)
BATCH_SIZE=100
DURATION=60  # 压测时长(秒)
PACKET_RATE=10000  # 每秒发送数据包

echo "Starting performance benchmark..."

for policy_count in "${POLICY_COUNTS[@]}"; do
    echo "Testing with $policy_count policies..."

    # 1. 生成 $policy_count 条策略
    ./migrate start --batch-size=$BATCH_SIZE --count=$policy_count

    # 2. 运行性能基准测试(发送 DURATION 秒数据)
    timeout $DURATION \
        ./load_generator \
            --rate=$PACKET_RATE \
            --policies=$policy_count \
            --output="results/benchmark_${policy_count}.json" \
        || true

    # 3. 收集 eBPF 统计
    bpftool map dump name policy_stats > "results/stats_${policy_count}.json"

    # 4. 清理策略
    ./migrate rollback
done

# 5. 生成性能报告
python3 generate_report.py --input=results/ --output=performance-benchmark-report.md
```

#### 性能数据收集 (Python)

```python
# generate_report.py - 性能报告生成

import json
import statistics
from pathlib import Path

class BenchmarkAnalyzer:
    def __init__(self, results_dir):
        self.results_dir = Path(results_dir)

    def analyze_latency(self, events_file):
        """分析查询延迟统计"""
        with open(events_file) as f:
            events = json.load(f)['events']

        latencies = [e['lookup_time_ns'] / 1000 for e in events]  # 转换为 μs

        return {
            'count': len(latencies),
            'min': min(latencies),
            'max': max(latencies),
            'mean': statistics.mean(latencies),
            'median': statistics.median(latencies),
            'p95': self.percentile(latencies, 95),
            'p99': self.percentile(latencies, 99),
            'stddev': statistics.stdev(latencies),
        }

    def percentile(self, data, p):
        """计算百分位数"""
        sorted_data = sorted(data)
        index = int(len(sorted_data) * p / 100)
        return sorted_data[index]

    def generate_report(self):
        """生成性能报告"""
        report = "# 性能基准报告\n\n"

        for policy_file in sorted(self.results_dir.glob('benchmark_*.json')):
            policy_count = policy_file.stem.split('_')[1]
            stats = self.analyze_latency(policy_file)

            report += f"## {policy_count} 条策略\n\n"
            report += f"- 查询总数: {stats['count']}\n"
            report += f"- 平均延迟: {stats['mean']:.2f} μs\n"
            report += f"- 中位延迟: {stats['median']:.2f} μs\n"
            report += f"- P99 延迟: {stats['p99']:.2f} μs\n"
            report += f"- 最大延迟: {stats['max']:.2f} μs\n"
            report += f"- 标准差: {stats['stddev']:.2f} μs\n\n"

        return report
```

#### 兼容性测试脚本

```bash
#!/bin/bash
# kernel_compat_test.sh

KERNELS=("4.11" "5.4" "5.10" "5.15")
POLICY_COUNT=1000

echo "Running kernel compatibility tests..."

for kernel in "${KERNELS[@]}"; do
    echo "Testing on Kernel $kernel..."

    # 1. 启动指定内核版本的 VM/容器
    docker run -v $(pwd):/workspace \
        ubuntu:kernel-$kernel \
        /bin/bash -c '
            cd /workspace

            # 2. 编译 eBPF 程序
            clang -O2 -c src/bpf/headers/policy_match_lpm.h -o policy.o

            # 3. 运行 verifier 检查
            bpftool gen object policy_verified.o policy.o
            bpftool gen skeleton policy_verified.o > policy.skel.h

            # 4. 加载 eBPF 程序
            sudo ./load_ebpf_program policy.skel.h

            # 5. 运行性能压测
            timeout 60 ./load_generator --policies=$POLICY_COUNT \
                --output=results/kernel_${kernel}.json

            # 6. 收集 verifier 日志
            bpftool prog show > results/verifier_${kernel}.log
        '
done

# 7. 对比分析
python3 analyze_compat.py results/
```
